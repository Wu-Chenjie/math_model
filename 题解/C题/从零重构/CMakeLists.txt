cmake_minimum_required(VERSION 3.20)
project(microgrid_from_zero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(HIGHS_ROOT "" CACHE PATH "HiGHS prefix or CasADi package directory")
add_executable(microgrid src/main.cpp)
add_executable(report src/report.cpp)
find_path(HIGHS_INCLUDE interfaces/highs_c_api.h HINTS "${HIGHS_ROOT}/include/highs" "${HIGHS_ROOT}/include")
find_library(HIGHS_LIBRARY NAMES highs HINTS "${HIGHS_ROOT}" "${HIGHS_ROOT}/lib")
if(HIGHS_INCLUDE AND HIGHS_LIBRARY)
  add_executable(verify src/verify.cpp)
  target_include_directories(verify PRIVATE "${HIGHS_INCLUDE}")
  target_link_libraries(verify PRIVATE "${HIGHS_LIBRARY}")
  get_filename_component(HIGHS_LIBDIR "${HIGHS_LIBRARY}" DIRECTORY)
  set_target_properties(verify PROPERTIES BUILD_RPATH "${HIGHS_LIBDIR}")
else()
  message(WARNING "Native HiGHS not found: independent LP verification target unavailable. Do not claim that test passed.")
endif()
enable_testing()
add_test(NAME analytic_gradient COMMAND microgrid gradient)
if(TARGET verify AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/inputs/native/data_load.bin")
  add_test(NAME independent_lp_and_causality COMMAND verify "${CMAKE_CURRENT_SOURCE_DIR}/inputs/native" "${CMAKE_CURRENT_BINARY_DIR}/verification.csv")
endif()
